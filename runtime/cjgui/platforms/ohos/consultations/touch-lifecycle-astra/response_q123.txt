   actual=20 expected=50（END.y−lastY 未入累计）。

   同绑定（节点/资源/动作/几何全同）场景版本=101，核心 resolveInput 严格比版本

    begin.recordIndex = static_cast<uint32_t>(liftIndex);
    begin.nodeId = node.pod.nodeId;
    begin.projectionVersion = s.gesture.targetProjectionVersion;
    begin.resourceId = node.pod.resourceId;
    begin.nodeKind = kind;
    begin.pointerX = static_cast<int64_t>(x);
    begin.pointerY = static_cast<int64_t>(y);
    s.events.push_back(begin);
    QueuedEvent end = begin;
    end.kind = kEvPointerEnd;
    end.pointerX = static_cast<int64_t>(x);
    end.pointerY = static_cast<int64_t>(y);
    s.events.push_back(end);
}

void synthesizeEventsFromRawTouch(Session &s, uint32_t action, float x, float y)
{
    // B 合成策略：单指手势，阈值前待定、超阈值由包含视口接管为连续滚动；
    // 有效抬起才执行一次点击。已激活编辑器保持长按/系统代理优先级，位移
    // 不强制当滚动。平台回调读取失败（queue 空/旧代拒绝）不会到达这里，
    // 也不合成合法触摸；坐标域与已接受场景节点一致。
    const int64_t nowMs = std::chrono::duration_cast<std::chrono::milliseconds>(
        std::chrono::steady_clock::now().time_since_epoch()).count();
    constexpr float kTouchScrollThresholdPx = 12.0f;

    if (action == CJGUI_OHOS_TOUCH_BEGIN) {
        // 单指身份：手势存续期间忽略新手指，不换绑、不重置累计位移。
        if (s.gesture.active) {
            RLOGI("touch begin ignored: gesture already active");
            return;
        }
        s.gesture = Session::TouchGesture{};
        s.gesture.active = true;
        s.gesture.startX = s.gesture.lastX = x;
        s.gesture.startY = s.gesture.lastY = y;
        s.gesture.surfaceGeneration = s.surfaceGeneration;
        size_t viewportIndex = 0;
        s.gesture.hasViewport = scrollAreaIndexContainingPoint(s, x, y, &viewportIndex);
        if (s.gesture.hasViewport) {
            s.gesture.viewportNodeId = s.accepted[viewportIndex].pod.nodeId;
            s.gesture.viewportResourceId = s.accepted[viewportIndex].pod.resourceId;
        }
        size_t index = 0;
        if (!hitTestAccepted(s, x, y, &index)) {
            return;  // 空白/非交互内容：无点击目标；滚动仍由包含视口接管。
        }
        const SceneNode &node = s.accepted[index];
        if (node.pod.isReadOnly != 0) return;  // 只读内容可滚动，不可激活
        const uint32_t kind = node.pod.nodeKind;
        const bool isEditableText = kind == kKindTextInput || kind == kKindIntegerInput;
        s.gesture.hasTarget = true;
        s.gesture.targetEditableText = isEditableText;
        s.gesture.targetNodeId = node.pod.nodeId;
        s.gesture.targetResourceId = node.pod.resourceId;
        s.gesture.targetNodeKind = kind;
        // B：起始绑定冻结——激活事件的场景版本取按下时刻；中途任何提交
        // （含同槽换绑动作/字段）都会使核心 resolveInput 拒绝旧按下激活，
        // 旧按下不能借新 sceneVersion 激活另一动作。
        s.gesture.targetProjectionVersion = node.pod.projectionVersion;
        if (isEditableText && s.editing && !s.editorRetired &&
            s.editingNodeId == node.pod.nodeId && s.editingResourceId == node.pod.resourceId) {
            // 已激活编辑器优先：caret 点击立即落账，长按计时开始；位移不
            // 接管为滚动（选区拖动/系统代理按明确优先级保留）。
            s.gesture.phase = Session::TouchGesture::kGestureEditorHold;
            s.gesture.pressBeginMs = nowMs;
            s.textPressBeginMs = nowMs;
            s.editingTapX = x - static_cast<double>(node.pod.x);
            s.editingTapPending = true;
        }
        return;
    }
    if (action == CJGUI_OHOS_TOUCH_UPDATE) {
        if (!s.gesture.active) return;  // 无手势的孤立位移不成合法触摸
        if (s.gesture.surfaceGeneration != s.surfaceGeneration) {
            // Surface 换代/退役：旧代手势不得延续到新挂载。
            cancelTouchGestureLocked(s);
            return;
        }
        const float dx = x - s.gesture.startX;
        const float dy = y - s.gesture.startY;
        // 差分样本必须在 last 更新前计算。待定期间样本不推进 last；
        // 视口接管后的首个差分即从起始点累计的全部位移，总量不丢。
        const float dySample = y - s.gesture.lastY;
        if (s.gesture.phase == Session::TouchGesture::kGesturePending) {
            if (std::fabs(dx) < kTouchScrollThresholdPx && std::fabs(dy) < kTouchScrollThresholdPx) {
                return;
            }
            if (s.gesture.hasViewport) {
                // 视口接管：取消子控件点击。纯滚动不结束编辑、不触发焦点
                // 切换结算，业务 owner 与草稿保持不动。
                s.gesture.phase = Session::TouchGesture::kGestureScroll;
                RLOGI("gesture scroll takeover viewport=%{public}llu target=%{public}llu",
                      static_cast<unsigned long long>(s.gesture.viewportNodeId),
                      static_cast<unsigned long long>(s.gesture.targetNodeId));
            } else if (s.gesture.hasTarget && !s.gesture.targetEditableText) {
                // 无包含视口：维持指针相位流（BEGIN 立即补发，保持序列）。
                s.gesture.phase = Session::TouchGesture::kGesturePointerDrag;
                size_t index = 0;
                if (sceneIndexByIdentityLocked(s, s.gesture.targetNodeId, s.gesture.targetResourceId,
                                               s.gesture.targetNodeKind, &index)) {
                    const SceneNode &node = s.accepted[index];
                    QueuedEvent ev;
                    ev.kind = kEvPointerBegin;
                    ev.recordIndex = static_cast<uint32_t>(index);
                    ev.nodeId = node.pod.nodeId;
                    ev.projectionVersion = node.pod.projectionVersion;
                    ev.resourceId = node.pod.resourceId;
                    ev.nodeKind = node.pod.nodeKind;
                    ev.pointerX = static_cast<int64_t>(x);
                    ev.pointerY = static_cast<int64_t>(y);
                    s.events.push_back(ev);
                    s.gesture.pointerStreamOpen = true;
                } else {
                    cancelTouchGestureLocked(s);  // 目标退役：无流可续
                    return;
                }
            } else {
                // 可编辑文本上起手且无包含视口：不滚动不拖动，待定到抬起。
                return;
            }
        }
        s.gesture.lastX = x;
        s.gesture.lastY = y;
        if (s.gesture.phase == Session::TouchGesture::kGestureScroll) {
            // 连续逻辑位移：共同 scroll 意图，不做半页跳转、不写布局坐标。
            // 手指下拖（dySample>0）露出上方内容 → offset 减小（delta 取负）。
            // 位移以浮点累计余量，只有整数部分交付共享整数 viewport：
            // 逐样本 float→Int64 截断会丢失小数位移，"by:0" 也不是滚动。
            if (dySample == 0.0f) return;
            size_t index = 0;
            if (!sceneIndexByIdentityLocked(s, s.gesture.viewportNodeId, s.gesture.viewportResourceId,
                                            CJGUI_INTERNAL_RENDERER_COMPOSABLE_SCROLL_AREA, &index)) {
                cancelTouchGestureLocked(s);  // 视口移除/换绑：手势取消
                return;
            }
            s.gesture.scrollAccumY += dySample;
            int64_t whole = static_cast<int64_t>(s.gesture.scrollAccumY);  // 截断保余量
            if (whole == 0) return;
            s.gesture.scrollAccumY -= static_cast<float>(whole);
            appendScrollIntentLocked(s, s.gesture.viewportNodeId, s.gesture.viewportResourceId,
                                     s.accepted[index].pod.projectionVersion, -whole);
            return;
        }
        if (s.gesture.phase == Session::TouchGesture::kGesturePointerDrag) {
            size_t index = 0;
            if (!sceneIndexByIdentityLocked(s, s.gesture.targetNodeId, s.gesture.targetResourceId,
                                            s.gesture.targetNodeKind, &index)) {
                cancelTouchGestureLocked(s);
                return;
            }
            const SceneNode &node = s.accepted[index];
            QueuedEvent ev;
            ev.kind = kEvPointerUpdate;
            ev.recordIndex = static_cast<uint32_t>(index);
            ev.nodeId = node.pod.nodeId;
            ev.projectionVersion = node.pod.projectionVersion;
            ev.resourceId = node.pod.resourceId;
            ev.nodeKind = node.pod.nodeKind;
            ev.pointerX = static_cast<int64_t>(x);
            ev.pointerY = static_cast<int64_t>(y);
            if (!s.events.empty()) {
                QueuedEvent &tail = s.events.back();
                if (tail.kind == kEvPointerUpdate && tail.nodeId == ev.nodeId) {
                    tail.pointerX = ev.pointerX;
                    tail.pointerY = ev.pointerY;
                    return;
                }
            }
            s.events.push_back(ev);
            return;
        }
        return;
    }
    if (action == CJGUI_OHOS_TOUCH_END || action == CJGUI_OHOS_TOUCH_CANCEL) {
        if (!s.gesture.active) return;
        // Surface 换代：旧代相位不得影响新挂载，等价取消。
        if (s.gesture.surfaceGeneration != s.surfaceGeneration) {
            cancelTouchGestureLocked(s);
            return;
        }
        if (action == CJGUI_OHOS_TOUCH_CANCEL) {
            cancelTouchGestureLocked(s);
            return;
        }
        const uint32_t phase = s.gesture.phase;
        switch (phase) {
            case Session::TouchGesture::kGesturePending:
                // B：END 重判阈值——快速轻扫（BEGIN 后无 UPDATE）的全部位移在
                // 抬起结算为一次滚动，不得当作点击激活；仅有界视口接管。
                if (s.gesture.hasViewport &&
                    (std::fabs(x - s.gesture.startX) >= kTouchScrollThresholdPx ||
                     std::fabs(y - s.gesture.startY) >= kTouchScrollThresholdPx)) {
                    size_t index = 0;
                    if (sceneIndexByIdentityLocked(s, s.gesture.viewportNodeId, s.gesture.viewportResourceId,
                                                   CJGUI_INTERNAL_RENDERER_COMPOSABLE_SCROLL_AREA, &index)) {
                        const int64_t whole = static_cast<int64_t>(y - s.gesture.startY);
                        if (whole != 0) {
                            appendScrollIntentLocked(s, s.gesture.viewportNodeId, s.gesture.viewportResourceId,
                                                     s.accepted[index].pod.projectionVersion, -whole);
                        }
                    }
                    break;
                }
                executePendingTapLocked(s, x, y, nowMs);
                break;
            case Session::TouchGesture::kGestureScroll: {
                // B：END 消费尾差——交付浮点累计余量的整数部分。
                size_t index = 0;
                if (sceneIndexByIdentityLocked(s, s.gesture.viewportNodeId, s.gesture.viewportResourceId,
                                               CJGUI_INTERNAL_RENDERER_COMPOSABLE_SCROLL_AREA, &index)) {
                    const int64_t whole = static_cast<int64_t>(s.gesture.scrollAccumY);
                    if (whole != 0) {
                        s.gesture.scrollAccumY -= static_cast<float>(whole);
                        appendScrollIntentLocked(s, s.gesture.viewportNodeId, s.gesture.viewportResourceId,
                                                 s.accepted[index].pod.projectionVersion, -whole);
                    }
                }
                break;
            }
            case Session::TouchGesture::kGesturePointerDrag: {
                size_t index = 0;
                if (sceneIndexByIdentityLocked(s, s.gesture.targetNodeId, s.gesture.targetResourceId,
                                               s.gesture.targetNodeKind, &index)) {
                    const SceneNode &node = s.accepted[index];
                    QueuedEvent ev;
                    ev.kind = kEvPointerEnd;
                    ev.recordIndex = static_cast<uint32_t>(index);
                    ev.nodeId = node.pod.nodeId;
                    ev.projectionVersion = node.pod.projectionVersion;
                    ev.resourceId = node.pod.resourceId;
                    ev.nodeKind = node.pod.nodeKind;
                    ev.pointerX = static_cast<int64_t>(x);
                    ev.pointerY = static_cast<int64_t>(y);
                    s.events.push_back(ev);
                    if (s.gesture.pointerStreamOpen) s.gesture.pointerStreamEnded = true;
                }
                break;
            }
            case Session::TouchGesture::kGestureEditorHold:
                // 已激活编辑器长按 ≥400ms → 全选（既有语义保持）。
                if (s.editing && !s.editorRetired && s.textPressBeginMs > 0 &&
                    nowMs - s.textPressBeginMs >= 400) {
                    s.selStartUtf16 = 0;
                    s.selEndUtf16 = static_cast<uint32_t>(s.editingText.size());
                    RLOGI("long-press selection: select-all len=%{public}u",
                          static_cast<unsigned>(s.selEndUtf16));
                    g_render.post(std::make_shared<RedrawJob>());
                }
                break;
        }
        cancelTouchGestureLocked(s);
        return;
    }
}

    private func isSplitSizeNavigation(text: String): Bool {
        return text == "left" || text == "right" || text == "up" || text == "down"
    }

    /// The keyboard size rule for a state-bound divider: the same ±8 nudge the
    /// handwritten split uses, bounded by the framework's ONE rule against the
    /// track the ACCEPTED scene really solved. Returns true when the size moved.
    private func applySplitStateKeyboardStep(node: CjguiComposableUiLayoutNode, text: String): Bool {
        if (let Some(state) <- splitStateForDivider(node)) {
            let direction = if (text == "left" || text == "up") {
                -1
            } else if (text == "right" || text == "down") {
                1
            } else {
                0
            }
            if (direction == 0) {
                return false
            }
        }
        return false
    }

    public func focusScopedNode(scope: CjguiComposableUiIdentityScope, localKey: String): Bool {
        let reference = CjguiComposableUiNodeReference.scoped(scope, localKey)
        let nodeId = identityRegistry.acceptedNodeIdForReference(reference)
        if (nodeId >= 0 && revealAcceptedNodeIfNeeded(nodeId)) {
            // The target exists but is outside the visible part of its scroll
            // viewport. Reveal it, remember the SAME request, and let the commit
            // that makes it visible satisfy the focus: keyboard navigation must
            // reach an editor below the fold instead of silently doing nothing.
            pendingScopedFocusScope = Some(scope)
            pendingScopedFocusKey = localKey
            requestRefresh()
            return false
        }
        if (nodeId < 0) {
            // The row may only exist once the scene that materializes it has
            // been accepted (an off-screen tree target that was just revealed).
            // Remember the request and retry right after the next commit
            // instead of dropping it and leaving the keyboard without a target.
            pendingScopedFocusScope = Some(scope)
            pendingScopedFocusKey = localKey
            return false
        }
        pendingScopedFocusScope = None
        pendingScopedFocusKey = ""
        return focusProjectedNode(nodeId)
    }

    private func isEditableTextNodeKind(nodeKind: Int64): Bool {
        return nodeKind == CJGUI_COMPOSABLE_UI_TEXT_INPUT || nodeKind == CJGUI_COMPOSABLE_UI_INTEGER_INPUT ||
            nodeKind == CJGUI_COMPOSABLE_UI_MULTILINE_TEXT_INPUT
    }
    /// Admit only the single-generation continuation that was produced by
    /// the live focused text adapter while its accepted successor scene was
    /// being assembled.  The current owner value must still be exactly the
    /// preceding local value; an external write, disable, delete or resource
    /// replacement fails one of these checks and remains rejectable.
    private func resolveLocalTextContinuation(eventKind: Int64, nodeId: Int64, resourceId: Int64, nodeKind: Int64,
        eventText: String, projectionVersion: Int64): CjguiComposableUiLayoutNode {
        if (eventKind != 28 || !isEditableTextNodeKind(nodeKind) || projectionVersion + 1 != nativeInputScene.version ||
            nodeId != localTextContinuationNodeId || resourceId != localTextContinuationResourceId || nodeKind !=
            localTextContinuationNodeKind || nodeId != focusedNodeId || resourceId != focusedResourceId || nodeKind !=
            focusedNodeKind || eventText == localTextContinuationValue) {
            return CjguiComposableUiLayoutNode.missing()
        }
        // `nativeInputScene` is a rendered input snapshot, not the real-time
        // domain owner.  In particular, an external same-object replacement
        // can leave its value unchanged while advancing the controller's
        // declared revision.  Admit this one callback only while the durable

**Q1｜压缩等价条件与最小充分保留集**

裁决：**“最新 MOVE＋曾越阈值”不足以保证等价；压缩必须保留不可逆的手势判决，以及夹紧运算的顺序。**

- 等价范围先限定为：同一手势、同一绑定与坐标域、viewport 上下界不变、途中没有外部滚动写入或其他可观察副作用。此时可保证手势分类、终态偏移与余量等价；不能据此宣称逐次回调、画面或时间行为等价。
- 当前差分语义下，最小充分的**有序样本骨架**为：`BEGIN → 首次跨阈值样本 → 后续每个 Y 方向转折点 → 最新样本 → END/CANCEL`。首次跨阈值须按现有 `abs(dx)>=12 || abs(dy)>=12` 判定，保留完整坐标；不能只存最大距离或布尔位。首次跨越之前的 MOVE，在当前 pending 不更新 last 的语义下可省略。END 坐标必须作为最后一次差分消费，然后终结；CANCEL 不补位移。
- 首次跨越与每个转折点均不可覆盖；仅同一单调段内部样本可折叠为末点。跨越事实一经成立，回到起点也不得恢复点击资格。转折数量没有固定上界，故该骨架仍可能满载，必须进入 Q2，不能继续删“冗余 MOVE”。
- **第二层队列也必须守同一规则。** 当前 `appendScrollIntentLocked` 无条件累加同版本 delta，会再次抹掉转折。只允许合并连续同号的整数滚动量，反号分段按序交给共享 viewport 夹紧；例如 offset=0 的 `-30,+30` 必须得到 30。
- 任意 pointer-drag 消费者可能依赖轨迹；不能套用 Y 转折压缩。bridge 尚不知道目标和阶段时，默认保留原始样本，除非已有可验证的分类信息。浮点重分组也不保证逐位相等；若要求严格整数输出等价，应保留原顺序量化结果，再合并同号整数段。

Flutter 的阈值锁存与 arena 一次性裁决支持“接受后不可回退”的状态设计，**不构成丢弃中间轨迹的等价证明**。

**Q2｜满载受害者与 CANCEL 位置**

裁决：**按 `(surfaceGeneration, gestureEpoch)` 整手势淘汰；禁止单独删除唯一 MOVE，禁止按 surface 整组清空。**

1. 每次有效 BEGIN 分配新的 `gestureEpoch`，贯穿 MOVE、END、CANCEL；手指 ID 重用不重用代次。记录 BEGIN 是否已经出队，包括已交给 renderer、尚未处理完的情况。
2. 压缩后仍满，选择**仍占队列空间的最老手势**作为受害者，尽量保住较新的完整手势。删除受害者全部未消费记录；若新到记录也属于受害者，一并丢弃，并抑制该代后续 MOVE/END，直到物理终结。不能留下 END 使其退化为点击。
3. BEGIN 尚未出队：整段删除即可，消费者没有状态需要取消。BEGIN 已出队或在途：必须投递恰好一次带受害者身份的 CANCEL，**逻辑位置在已出队前缀之后、被删除后缀的起点，且先于后继手势 BEGIN**。不能无条件插全局队首，也不能放到新手势之后。
4. 预留取消控制容量，并将它计入总容量；取消标记不得再被当成受害数据淘汰。空间仍不足，继续淘汰其他尚未开始的整手势；不要反复删除、重插同一个 CANCEL。
5. 出队、淘汰与取消交付必须形成确定顺序；renderer/core 只终结身份匹配的手势。**未解决旧 CANCEL 对新捕获的无身份反馈之前，任何队列淘汰方案都不能算闭合。**

过载取消是明确的降级终态，不是无损压缩，也不能撤回已经发生的滚动。

**Q3｜绑定身份两条路与推荐**

裁决：**推荐新增核心签发的 `bindingEpoch`，贯穿 node POD 和事件 POD；`projectionVersion` 继续只表示场景版本。**

- **核心二次校验路线可以成立，但必须有历史证据。** 利用事件冻结的旧 `projectionVersion` 找到按下时的绑定快照，再与 flush 后当前绑定比较。只查“当前节点存在＋当前绑定哈希”无法知道按下时绑定了谁。快照须覆盖 `node/resource/kind、semanticId/incarnation、fieldId、actionName、operationActionName、operationResourceId` 等身份；还须识别移除重建、A→B→A，不能仅比较最终字段值。哈希只能辅助，不能代替完整身份判等。此路节省 POD 布局变更，却增加历史保留、在途租约及回收契约；固定保存最近几帧会再次误杀长按或排队事件。
- **`bindingEpoch` 路线更直接。** 核心在真实换绑、退役重建时递增；同绑定重绘、换帧、值或几何更新不因此递增。已有 `semanticIncarnation` 只有在确认覆盖全部绑定变更后才可复用。renderer 在 BEGIN 冻结 epoch，后续重验及事件回传使用该值；核心执行前再次比较，并检查当前可输入性、模态和适用的命中条件。目标与 viewport 分别保存自身绑定身份。
- 代价是 C/Cangjie 布局、snapshot、复制与事件传输路径、ABI 校验必须同步；**只给 node POD 加字段而不让事件携带，无法闭环。** 未携带 epoch 的旧事件应保留原严格版本规则，不能把零值当通配符。
- 推荐此路还有一个现实理由：`executePendingTapLocked` 会在核心收到事件前结束编辑或启动新编辑上下文。仅靠核心事后拒绝，无法阻止这些错误副作用；renderer 持有 epoch 后可在副作用发生前拒绝换绑。

snapshot 的 pending focus 应同时补齐主核心现有的完整绑定比较，包含 `operationResourceId`；新增 epoch 不应成为漏校验的理由。
